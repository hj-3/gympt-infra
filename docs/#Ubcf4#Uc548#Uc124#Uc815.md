## 네트워크 보안

### VPC 격리

**서브넷 분리**:
```
Public Subnet (인터넷 접근)
└── ALB, NAT Gateway만 배치

Private Subnet (NAT Gateway 경유)
└── EKS Nodes, Lambda 함수

Database Subnet (완전 격리)
└── RDS, ElastiCache (인터넷 접근 불가)
```

**보안 그룹 구성**:

```hcl
# ALB Security Group
resource "aws_security_group" "alb" {
  name        = "gympt-${var.environment}-alb-sg"
  vpc_id      = aws_vpc.main.id

  # HTTPS만 허용
  ingress {
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
    description = "Allow HTTPS from internet"
  }

  # HTTP (CloudFront에서만)
  ingress {
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
    description = "Allow HTTP for redirect"
  }

  # EKS NodePort로만 Egress
  egress {
    from_port       = 30000
    to_port         = 32767
    protocol        = "tcp"
    security_groups = [aws_security_group.eks_nodes.id]
    description     = "Allow traffic to EKS nodes"
  }

  tags = {
    Name = "gympt-${var.environment}-alb-sg"
  }
}

# EKS Node Security Group
resource "aws_security_group" "eks_nodes" {
  name        = "gympt-${var.environment}-eks-nodes-sg"
  vpc_id      = aws_vpc.main.id

  # ALB에서만 접근
  ingress {
    from_port       = 30000
    to_port         = 32767
    protocol        = "tcp"
    security_groups = [aws_security_group.alb.id]
    description     = "Allow NodePort from ALB"
  }

  # 노드 간 통신
  ingress {
    from_port = 0
    to_port   = 65535
    protocol  = "-1"
    self      = true
    description = "Allow inter-node communication"
  }

  # 모든 Egress (VPC Endpoints 접근 포함)
  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
    description = "Allow all outbound"
  }

  tags = {
    Name = "gympt-${var.environment}-eks-nodes-sg"
  }
}

# RDS Security Group
resource "aws_security_group" "rds" {
  name        = "gympt-${var.environment}-rds-sg"
  vpc_id      = aws_vpc.main.id

  # EKS 노드에서만 접근
  ingress {
    from_port       = 5432
    to_port         = 5432
    protocol        = "tcp"
    security_groups = [aws_security_group.eks_nodes.id]
    description     = "Allow PostgreSQL from EKS nodes"
  }

  # Egress 없음 (응답만)
  egress {
    from_port       = 0
    to_port         = 0
    protocol        = "-1"
    security_groups = [aws_security_group.eks_nodes.id]
    description     = "Response traffic only"
  }

  tags = {
    Name = "gympt-${var.environment}-rds-sg"
  }
}

# ElastiCache Security Group
resource "aws_security_group" "redis" {
  name        = "gympt-${var.environment}-redis-sg"
  vpc_id      = aws_vpc.main.id

  # EKS 노드에서만 접근
  ingress {
    from_port       = 6379
    to_port         = 6379
    protocol        = "tcp"
    security_groups = [aws_security_group.eks_nodes.id]
    description     = "Allow Redis from EKS nodes"
  }

  tags = {
    Name = "gympt-${var.environment}-redis-sg"
  }
}
```

### Network ACLs

```hcl
# Public Subnet NACL
resource "aws_network_acl" "public" {
  vpc_id     = aws_vpc.main.id
  subnet_ids = aws_subnet.public[*].id

  # Inbound: HTTP/HTTPS 허용
  ingress {
    rule_no    = 100
    protocol   = "tcp"
    from_port  = 443
    to_port    = 443
    cidr_block = "0.0.0.0/0"
    action     = "allow"
  }

  ingress {
    rule_no    = 110
    protocol   = "tcp"
    from_port  = 80
    to_port    = 80
    cidr_block = "0.0.0.0/0"
    action     = "allow"
  }

  # Ephemeral ports (응답 트래픽)
  ingress {
    rule_no    = 200
    protocol   = "tcp"
    from_port  = 1024
    to_port    = 65535
    cidr_block = "0.0.0.0/0"
    action     = "allow"
  }

  # Outbound: 모두 허용
  egress {
    rule_no    = 100
    protocol   = "-1"
    from_port  = 0
    to_port    = 0
    cidr_block = "0.0.0.0/0"
    action     = "allow"
  }

  tags = {
    Name = "gympt-${var.environment}-public-nacl"
  }
}

# Private Subnet NACL
resource "aws_network_acl" "private" {
  vpc_id     = aws_vpc.main.id
  subnet_ids = aws_subnet.private[*].id

  # VPC 내부 트래픽만 허용
  ingress {
    rule_no    = 100
    protocol   = "-1"
    from_port  = 0
    to_port    = 0
    cidr_block = var.vpc_cidr
    action     = "allow"
  }

  egress {
    rule_no    = 100
    protocol   = "-1"
    from_port  = 0
    to_port    = 0
    cidr_block = "0.0.0.0/0"
    action     = "allow"
  }

  tags = {
    Name = "gympt-${var.environment}-private-nacl"
  }
}
```

## IAM 및 인증

### IRSA (IAM Roles for Service Accounts)

**EKS OIDC Provider 설정**:

```hcl
# OIDC Provider
data "tls_certificate" "eks" {
  url = aws_eks_cluster.main.identity[0].oidc[0].issuer
}

resource "aws_iam_openid_connect_provider" "eks" {
  url = aws_eks_cluster.main.identity[0].oidc[0].issuer

  client_id_list = ["sts.amazonaws.com"]

  thumbprint_list = [
    data.tls_certificate.eks.certificates[0].sha1_fingerprint
  ]

  tags = {
    Name = "gympt-${var.environment}-eks-oidc"
  }
}
```

**Service Account IAM Role 예시**:

```hcl
# Backend API IAM Role
resource "aws_iam_role" "backend_api" {
  name = "gympt-${var.environment}-backend-api"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect = "Allow"
      Principal = {
        Federated = aws_iam_openid_connect_provider.eks.arn
      }
      Action = "sts:AssumeRoleWithWebIdentity"
      Condition = {
        StringEquals = {
          "${replace(aws_eks_cluster.main.identity[0].oidc[0].issuer, "https://", "")}:sub" = "system:serviceaccount:backend-api:backend-api-sa"
          "${replace(aws_eks_cluster.main.identity[0].oidc[0].issuer, "https://", "")}:aud" = "sts.amazonaws.com"
        }
      }
    }]
  })
}

# S3 액세스 정책
```

### MFA 강제

```hcl
# MFA가 활성화되지 않은 사용자의 액세스 거부
resource "aws_iam_policy" "require_mfa" {
  name = "gympt-require-mfa"

  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Sid    = "DenyAllExceptListedIfNoMFA"
      Effect = "Deny"
      NotAction = [
        "iam:CreateVirtualMFADevice",
        "iam:EnableMFADevice",
        "iam:GetUser",
        "iam:ListMFADevices",
        "iam:ListVirtualMFADevices",
        "iam:ResyncMFADevice",
        "sts:GetSessionToken"
      ]
      Resource = "*"
      Condition = {
        BoolIfExists = {
          "aws:MultiFactorAuthPresent" = "false"
        }
      }
    }]
  })
}
```

```hcl
  rotation_rules {
    automatically_after_days = 30
  }

```

**Application에서 Secrets 접근**:

```python
import boto3
import json

def get_secret(secret_name):
    client = boto3.client('secretsmanager', region_name='ap-northeast-2')
    
    response = client.get_secret_value(SecretId=secret_name)
    secret = json.loads(response['SecretString'])
    
    return secret

# 사용 예시
rds_creds = get_secret('gympt/dev/rds-password')
username = rds_creds['username']
password = rds_creds['password']
```

## 모니터링 및 감사

### CloudTrail 설정

```hcl
# CloudTrail S3 Bucket
resource "aws_s3_bucket" "cloudtrail" {
  bucket = "gympt-cloudtrail-${data.aws_caller_identity.current.account_id}"
}

resource "aws_s3_bucket_lifecycle_configuration" "cloudtrail" {
  bucket = aws_s3_bucket.cloudtrail.id

  rule {
    id     = "archive-old-logs"
    status = "Enabled"

    transition {
      days          = 90
      storage_class = "GLACIER"
    }

    expiration {
      days = 365
    }
  }
}

# CloudTrail
resource "aws_cloudtrail" "main" {
  name                          = "gympt-${var.environment}-trail"
  s3_bucket_name                = aws_s3_bucket.cloudtrail.id
  include_global_service_events = true
  is_multi_region_trail         = true
  enable_log_file_validation    = true

  event_selector {
    read_write_type           = "All"
    include_management_events = true

    data_resource {
      type   = "AWS::S3::Object"
      values = ["arn:aws:s3:::gympt-*/*"]
    }
  }

  cloud_watch_logs_group_arn = "${aws_cloudwatch_log_group.cloudtrail.arn}:*"
  cloud_watch_logs_role_arn  = aws_iam_role.cloudtrail_cloudwatch.arn

  tags = {
    Name        = "gympt-${var.environment}-trail"
    Environment = var.environment
  }
}
```

### 보안 알람

```hcl
# 루트 계정 사용 알람
resource "aws_cloudwatch_log_metric_filter" "root_account_usage" {
  name           = "gympt-root-account-usage"
  log_group_name = aws_cloudwatch_log_group.cloudtrail.name

  pattern = <<PATTERN
{
  $.userIdentity.type = "Root" &&
  $.userIdentity.invokedBy NOT EXISTS &&
  $.eventType != "AwsServiceEvent"
}
PATTERN

  metric_transformation {
    name      = "RootAccountUsageCount"
    namespace = "GYMPT/Security"
    value     = "1"
  }
}

resource "aws_cloudwatch_metric_alarm" "root_account_usage" {
  alarm_name          = "gympt-${var.environment}-root-account-usage"
  comparison_operator = "GreaterThanThreshold"
  evaluation_periods  = 1
  metric_name         = "RootAccountUsageCount"
  namespace           = "GYMPT/Security"
  period              = 60
  statistic           = "Sum"
  threshold           = 0
  alarm_description   = "Root account was used"
  alarm_actions       = [aws_sns_topic.security_alerts.arn]
}

# IAM 정책 변경 알람
resource "aws_cloudwatch_log_metric_filter" "iam_policy_changes" {
  name           = "gympt-iam-policy-changes"
  log_group_name = aws_cloudwatch_log_group.cloudtrail.name

  pattern = <<PATTERN
{
  ($.eventName = DeleteGroupPolicy) ||
  ($.eventName = DeleteRolePolicy) ||
  ($.eventName = DeleteUserPolicy) ||
  ($.eventName = PutGroupPolicy) ||
  ($.eventName = PutRolePolicy) ||
  ($.eventName = PutUserPolicy) ||
  ($.eventName = CreatePolicy) ||
  ($.eventName = DeletePolicy) ||
  ($.eventName = CreatePolicyVersion) ||
  ($.eventName = DeletePolicyVersion) ||
  ($.eventName = AttachRolePolicy) ||
  ($.eventName = DetachRolePolicy) ||
  ($.eventName = AttachUserPolicy) ||
  ($.eventName = DetachUserPolicy) ||
  ($.eventName = AttachGroupPolicy) ||
  ($.eventName = DetachGroupPolicy)
}
PATTERN

  metric_transformation {
    name      = "IAMPolicyChangeCount"
    namespace = "GYMPT/Security"
    value     = "1"
  }
}

# Security Group 변경 알람
resource "aws_cloudwatch_log_metric_filter" "security_group_changes" {
  name           = "gympt-security-group-changes"

  pattern = <<PATTERN
{
  ($.eventName = AuthorizeSecurityGroupIngress) ||
  ($.eventName = AuthorizeSecurityGroupEgress) ||
  ($.eventName = RevokeSecurityGroupIngress) ||
  ($.eventName = RevokeSecurityGroupEgress) ||
  ($.eventName = CreateSecurityGroup) ||
  ($.eventName = DeleteSecurityGroup)
```
